# /etc/systemd/system/unicon-master.service
[Unit]
Description=UNICON Master (cam+luna+imu+etc)
After=network.target

[Service]
Type=simple
User=pi
WorkingDirectory=/home/pi/UNICON
ExecStart=/usr/bin/python3 /home/pi/UNICON/src/unicon_run.py
Restart=always
RestartSec=2
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
